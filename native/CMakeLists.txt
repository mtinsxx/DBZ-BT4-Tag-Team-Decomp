cmake_minimum_required(VERSION 3.22)
project(tenkaichi_native_engine LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(tenkaichi_assets STATIC src/portrait.cpp src/iso_content.cpp src/sha256.cpp src/preparation.cpp src/inspection.cpp
    ../android/src/bt4_iso_source.c ../android/src/bt4_iso9660.c ../port/src/bt4_base_profile.c)
target_include_directories(tenkaichi_assets PUBLIC include PRIVATE ../android/include ../port/src ../port/include)
target_link_libraries(tenkaichi_assets PUBLIC Threads::Threads)
set_target_properties(tenkaichi_assets PROPERTIES POSITION_INDEPENDENT_CODE ON
    C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_compile_options(tenkaichi_assets PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing -ffunction-sections -fdata-sections)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "This native engine target requires arm64-v8a")
    endif()
    find_package(SDL3 3.4.12 EXACT REQUIRED CONFIG)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/resource_shaders.hpp"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/shaders.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "${CMAKE_CURRENT_BINARY_DIR}/generated/resource_shaders.hpp"
        DEPENDS shaders/resource.vert shaders/resource.frag tools/shaders.py VERBATIM)
    add_library(tenkaichi_native SHARED src/android_entry.cpp src/renderer.cpp src/preparation_jni.cpp
        "${CMAKE_CURRENT_BINARY_DIR}/generated/resource_shaders.hpp")
    target_link_libraries(tenkaichi_native PRIVATE tenkaichi_assets SDL3::SDL3 android log)
    target_include_directories(tenkaichi_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ../android/include)
    target_compile_options(tenkaichi_native PRIVATE -Wall -Wextra -Werror)
    target_link_options(tenkaichi_native PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,-z,max-page-size=16384)
else()
    enable_testing()
    option(NATIVE_SANITIZERS "ASan/UBSan for native resource parsing" OFF)
    if(NATIVE_SANITIZERS)
        target_compile_options(tenkaichi_assets PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(tenkaichi_assets PUBLIC -fsanitize=address,undefined)
    endif()
    add_executable(native_assets_tests tests/assets_test.cpp)
    target_link_libraries(native_assets_tests PRIVATE tenkaichi_assets)
    target_include_directories(native_assets_tests PRIVATE ../android/include ../port/src ../port/include)
    target_compile_options(native_assets_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_executable(native_preparation_tests tests/preparation_test.cpp)
    target_link_libraries(native_preparation_tests PRIVATE tenkaichi_assets)
    target_include_directories(native_preparation_tests PRIVATE ../android/include ../port/src ../port/include)
    target_compile_options(native_preparation_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_portrait_reference.cpp"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/desktop_portrait_reference.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/../port/src/gs/fight_panels_ui.cpp"
            "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_portrait_reference.cpp"
        DEPENDS tools/desktop_portrait_reference.py ../port/src/gs/fight_panels_ui.cpp VERBATIM)
    add_executable(native_pixel_reference_tests tests/pixel_reference_test.cpp
        "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_portrait_reference.cpp")
    target_link_libraries(native_pixel_reference_tests PRIVATE tenkaichi_assets)
    target_compile_options(native_pixel_reference_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME native_desktop_pixel_reference COMMAND native_pixel_reference_tests)
    add_executable(native_presentation_tests tests/presentation_test.cpp)
    target_link_libraries(native_presentation_tests PRIVATE tenkaichi_assets)
    target_compile_options(native_presentation_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME native_presentation_contracts COMMAND native_presentation_tests)
    add_test(NAME native_preparation_contracts COMMAND native_preparation_tests)
    add_test(NAME native_resource_contracts COMMAND native_assets_tests)
    set_tests_properties(native_resource_contracts native_preparation_contracts native_presentation_contracts native_desktop_pixel_reference PROPERTIES TIMEOUT 30 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endif()
